/**
 * 核心处理函数
 * @param {Array} data - 输入数据
 * @param {Object} config - 配置选项
 * @returns {Array} 处理后的结果
 */
function processData(data, config = {}) {
  // 参数校验：确保输入符合预期
  if (!Array.isArray(data)) {
    throw new TypeError("data 必须是数组");
  }

  // 核心处理逻辑
  return data
    .filter(item => item != null)  // 过滤无效值
    .map(item => ({
      ...item,
      processed: true,
      timestamp: Date.now()
    }));
}
